// ================= ================= ================= // 1. خريطة معرفات ملفات Google Sheets الخاصة بكل شركة (16 شركة) // ================= ================= ================= var COMPANY_SHEETS = { "company_1": "ID_GOOGLE_SHEET_FOR_COMPANY_1", "company_2": "ID_GOOGLE_SHEET_FOR_COMPANY_2", "company_3": "ID_GOOGLE_SHEET_FOR_COMPANY_3", "company_4": "ID_GOOGLE_SHEET_FOR_COMPANY_4", "company_5": "ID_GOOGLE_SHEET_FOR_COMPANY_5", "company_6": "ID_GOOGLE_SHEET_FOR_COMPANY_6", "company_7": "ID_GOOGLE_SHEET_FOR_COMPANY_7", "company_8": "ID_GOOGLE_SHEET_FOR_COMPANY_8", "company_9": "ID_GOOGLE_SHEET_FOR_COMPANY_9", "company_10": "ID_GOOGLE_SHEET_FOR_COMPANY_10", "company_11": "ID_GOOGLE_SHEET_FOR_COMPANY_11", "company_12": "ID_GOOGLE_SHEET_FOR_COMPANY_12", "company_13": "ID_GOOGLE_SHEET_FOR_COMPANY_13", "company_14": "ID_GOOGLE_SHEET_FOR_COMPANY_14", "company_15": "ID_GOOGLE_SHEET_FOR_COMPANY_15", "company_16": "ID_GOOGLE_SHEET_FOR_COMPANY_16" }; // ================= ================= ================= // 2. دالة جلب البيانات (doGet) // ================= ================= ================= function doGet(e) { var companyId = e.parameter.companyId || "company_1"; var sheetName = e.parameter.sheet || "Sheet2"; var ssId = COMPANY_SHEETS[companyId]; if (!ssId) { return ContentService.createTextOutput(JSON.stringify({ "error": "Company ID not found" })) .setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.openById(ssId); var sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getDisplayValues(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ================= ================= ================= // 3. دالة إضافة وتحديث الحجوزات (doPost) // ================= ================= ================= function doPost(e) { try { var contents = JSON.parse(e.postData.contents); var companyId = contents.companyId || "company_1"; var ssId = COMPANY_SHEETS[companyId]; if (!ssId) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Invalid Company ID" })) .setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.openById(ssId); // أ. تحديث حالة الحجز (Accepted / Cancelled) if (contents.action === "updateStatus") { var sheet2 = ss.getSheetByName("Sheet2"); if (!sheet2) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Sheet2 not found" })) .setMimeType(ContentService.MimeType.JSON); } var data = sheet2.getDataRange().getDisplayValues(); var voucherToFind = String(contents.voucherNo).trim(); var updated = false; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === voucherToFind) { sheet2.getRange(i + 1, 15).setValue(contents.status); // العمود 15 (Status) updated = true; break; } } if (updated) { return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Status updated successfully" })) .setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Voucher not found" })) .setMimeType(ContentService.MimeType.JSON); } } // ب. إضافة حجز جديد if (contents.action === "addBooking" || !contents.action) { var sheet2 = ss.getSheetByName("Sheet2"); if (!sheet2) { sheet2 = ss.insertSheet("Sheet2"); sheet2.appendRow(["Voucher", "Client ID", "Customer Name", "Nationality", "Activity", "Supplier", "Guide", "Guests", "Date", "Pickup Time", "Payment Method", "Total", "Paid", "Notes", "Status"]); } sheet2.appendRow([ contents.voucherNo || "", contents.clientID || "", contents.customerName || "", contents.nationality || "", contents.activity || "", contents.supplier || "", contents.guide || "", contents.guests || "", contents.bookingDate || "", "'" + (contents.pickupTime || ""), contents.paymentMethod || "", contents.totalAmount || "", contents.paidAmount || "", contents.notes || "", "Pending" ]); return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Booking saved" })) .setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
